rules_version = '2';
// Reglas de seguridad de Grupo Los Panchos – pedidos directos
service cloud.firestore {
  match /databases/{database}/documents {

    function sesion() { return request.auth != null && request.auth.token.email_verified == true; }
    function correo() { return request.auth.token.email.lower(); }
    function esDueno() { return sesion() && correo() == '__CORREO_DEL_DUENO__'; }
    function esAdmin() { return esDueno() || (sesion() && exists(/databases/$(database)/documents/admins/$(correo()))); }

    // Horario, avisos y menú: los ve cualquiera, solo el equipo los cambia
    match /ajustes/{id} {
      allow read: if true;
      allow write: if esAdmin();
    }
    match /fotos/{id} {
      allow read: if true;
      allow write: if esAdmin();
    }
    match /admins/{mail} {
      allow read: if esAdmin();
      allow write: if esDueno();
    }

    // Pedidos completos (nombre, teléfono, dirección): solo el equipo los ve
    match /pedidos/{id} {
      allow create: if esAdmin() || (
        id.size() >= 16 && id.size() <= 40
        && request.resource.data.keys().hasOnly(['folio','ts','estado','modo','items','subtotal','envio','km','total',
             'nombre','tel','calle','colonia','refs','gps','hora','pago','cambio','notas','promos','regalo','sucursal','canal','v'])
        && request.resource.data.estado == 'nuevo'
        && request.resource.data.ts == request.time
        && request.resource.data.canal == 'web'
        && request.resource.data.items is list
        && request.resource.data.items.size() > 0 && request.resource.data.items.size() <= 80
        && request.resource.data.total is number && request.resource.data.total >= 0 && request.resource.data.total < 50000
        && request.resource.data.nombre is string && request.resource.data.nombre.size() <= 80
        && request.resource.data.tel is string && request.resource.data.tel.size() <= 20
      );
      allow read, update, delete: if esAdmin();
    }

    // Estado público del pedido (sin datos personales): el cliente lo sigue con su enlace
    match /estados/{id} {
      allow get: if true;
      allow list: if esAdmin();
      allow create: if esAdmin() || (
        id.size() >= 16 && id.size() <= 40
        && request.resource.data.keys().hasOnly(['folio','estado','modo','ts'])
        && request.resource.data.estado == 'nuevo'
        && request.resource.data.ts == request.time
      );
      allow update, delete: if esAdmin();
    }
  }
}
